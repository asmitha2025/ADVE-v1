ADVE Python Pip Library

Run production-grade local video embeddings directly on your edge hardware without cloud subscription fees.

# Install library via pip
pip install adve

# Index any local video file
from adve import VideoIndex

index = VideoIndex()
index.add("cctv_footage.mp4")

# Perform semantic search
results = index.search("person in a red hoodie walking near the gate")
# -> [{"timestamp": "02:14.30", "score": 0.948, "source": "visual"}]

# Run conversational Q&A
answer = index.ask("What time does the red hoodie person enter?")
# -> "The person in the red hoodie enters the gate at 02:14."

# 2. Query moments
