ADVE as a pip library
Like llama.cpp or whisper.cpp — run production video AI on your own hardware. No cloud. No API fees. Full control.
# install
pip install adve

# index any video
from adve import VideoIndex

index = VideoIndex()
index.add("lecture.mp4")

# semantic search
results = index.search("person explaining backpropagation")
# → [{"timestamp": "04:23", "score": 0.94, "source": "visual+audio"}]

# ask questions
answer = index.ask("When does the formula first appear?")
# → "The formula ∂L/∂w appears at 04:23 on the whiteboard."
